/** Gradient dV/dx **/
void dVdx(typeRNum *out, ctypeRNum T, ctypeRNum *x, ctypeRNum *p, ctypeRNum *xdes, typeUSERPARAM *userparam)
{
    ctypeRNum *pCost = (ctypeRNum*)userparam;      // Cost parameter pointer
    ctypeRNum *pTraj = (ctypeRNum*)userparam + N_PCOST; // Trajectory parameter pointer
    ctypeRNum *controlType = (ctypeRNum*)userparam + N_PCOST + N_PTRAJ; // Control type pointer
    ctypeRNum *t_ext = (ctypeRNum*)userparam + N_PCOST + N_PTRAJ + 1;   // Extended time

    typeRNum* S = (typeRNum*)userparam + 11; // Terminal weighting matrix
    typeRNum temp = 0;
    
   //7>>0
   //8>>1
   //9>>2
   //10>>3
    for (typeInt i = 0; i < 8; ++i)
    {
        if (i == 0) // Derivative for xB
        {
            // Trajectory contribution for xB
            temp = pTraj[0] * sin(pTraj[2] * pTraj[3] * (t_ext[0]+t));
            out[i] = S[i] * (x[i] - temp);
        }
        else if (i == 4) // Derivative for yB
        {
            // Trajectory contribution for yB
            temp = pTraj[1] * cos(pTraj[3] * (t_ext[0]+t));
            out[i] = S[i] * (x[i] - temp);
        }
        else if (i == 2) // Derivative for θ (tilt along x-axis)
        {
            // Contribution from ball-on-plate dynamics for θ
            typeRNum dQ_theta = pTraj[0] * cos(pTraj[2] * pTraj[3] * (t_ext[0]+t)); // Adjust based on dynamics
            out[i] = S[i] * dQ_theta * (x[i] - xdes[i]);
        }
        else if (i == 6) // Derivative for φ (tilt along y-axis)
        {
            // Contribution from ball-on-plate dynamics for φ
            typeRNum dQ_phi = -pTraj[1] * sin(pTraj[3] * (t_ext[0]+t)); // Adjust based on dynamics
            out[i] = S[i] * dQ_phi * (x[i] - xdes[i]);
        }
        else
        {
            // Standard quadratic term for other states
            out[i] = S[i] * (x[i] - xdes[i]);
        }
    }
}
